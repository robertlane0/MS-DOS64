# Bochs 3.0 x86-64 BIOS boot config for MS-DOS64
# See docs/05-boot-and-testing-strategy.md — updated for Bochs 3.0 model presets
megs: 256
romimage: file=$BXSHARE/BIOS-bochs-latest
vgaromimage: file=$BXSHARE/VGABIOS-lgpl-latest.bin
# CHS 20/16/63 matches the MBR/stage2 CHS-fallback assumptions (SPT=63,
# HPC=16). The 10 MiB image is 20480 sectors; CHS covers 20160 of them
# ("extra data outside CHS" warning is harmless: the kernel uses LBA
# exclusively after boot, and all used LBAs — kernel 16+, scratch
# 200/500-511, FAT12 volume 512-3391 — are < 20160).
ata0-master: type=disk, path="build/dos64.img", mode=flat, cylinders=20, heads=16, spt=63
boot: disk
log: bochs.log
cpu: model=ryzen, count=1, ips=50000000, reset_on_triple_fault=1, ignore_bad_msrs=1
panic: action=report
magic_break: enabled=1
# Serial is write-only capture (guest -> serial.log). The COMMAND64 shell
# prints banner/prompt here; interactive input comes from PS/2 (GUI) —
# for pipe-driven automation use QEMU -serial stdio instead.
# NOTE: the shell must use blocking THR waits, not drop-if-busy: Bochs
# emulates 16550 baud timing, so bursts (banner/prompt) lose bytes with
# non-blocking emits while QEMU (instant THRE) never drops.
com1: enabled=1, mode=file, dev=serial.log
display_library: nogui
